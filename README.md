Aplicación Full Stack para administrar clientes con:
- Frontend: HTML5, CSS3 y JavaScript (Fetch API).
- Backend: ASP.NET Core Web API con .NET 10 / Visual Studio 2026.
- Base de datos: Oracle Database 19c.
- Persistencia: procedimientos almacenados PL/SQL invocados con
  ODP.NET.
- Seguridad: autenticación JWT y contraseñas con PBKDF2 (nunca se
  almacenan en texto plano).
- Operaciones: CRUD de clientes y autenticación.

Ajusta la versión de .NET al SDK instalado. El ejemplo usa net10.0.
No publiques credenciales ni claves JWT en repositorios. Para
producción usa HTTPS, secretos administrados, políticas CORS
restrictivas, rotación de claves, rate limiting y auditoría.

CREATE OR REPLACE PACKAGE BODY pkg_seguridad AS
  PROCEDURE validar_usuario(
    p_nombre_usuario IN VARCHAR2,
    p_password_hash IN VARCHAR2,
    p_id_usuario OUT NUMBER,
    p_nombre OUT VARCHAR2,
    p_estado OUT NUMBER,
    p_mensaje OUT VARCHAR2
  ) AS
  BEGIN
    p_id_usuario := NULL;
    p_nombre := NULL;

    SELECT id_usuario, nombre_usuario
      INTO p_id_usuario, p_nombre
      FROM usuarios
     WHERE UPPER(nombre_usuario) = UPPER(p_nombre_usuario)
       AND password_hash = p_password_hash
       AND activo = 'S';

    p_estado := 1;
    p_mensaje := 'Autenticación correcta';
  EXCEPTION
    WHEN NO_DATA_FOUND THEN
      p_estado := 0;
      p_mensaje := 'Credenciales inválidas';
  END validar_usuario;
END pkg_seguridad;

CREATE OR REPLACE PACKAGE pkg_clientes AS
  PROCEDURE listar(p_resultado OUT SYS_REFCURSOR);
  PROCEDURE obtener(p_id IN NUMBER, p_resultado OUT SYS_REFCURSOR);
  PROCEDURE insertar(
    p_nombres IN VARCHAR2, p_apellidos IN VARCHAR2,
    p_correo IN VARCHAR2, p_telefono IN VARCHAR2,
    p_id OUT NUMBER
  );

PROCEDURE actualizar(
    p_id IN NUMBER, p_nombres IN VARCHAR2, p_apellidos IN VARCHAR2,
    p_correo IN VARCHAR2, p_telefono IN VARCHAR2, p_filas OUT NUMBER
  );

PROCEDURE eliminar(p_id IN NUMBER, p_filas OUT NUMBER);
END pkg_clientes;

CREATE OR REPLACE PACKAGE BODY pkg_clientes AS
  PROCEDURE listar(p_resultado OUT SYS_REFCURSOR) AS
  BEGIN
    OPEN p_resultado FOR
      SELECT id_cliente, nombres, apellidos, correo, telefono, fecha_registro
        FROM clientes ORDER BY id_cliente DESC;
  END;

  PROCEDURE obtener(p_id IN NUMBER, p_resultado OUT SYS_REFCURSOR) AS
  BEGIN
    OPEN p_resultado FOR
      SELECT id_cliente, nombres, apellidos, correo, telefono, fecha_registro
        FROM clientes WHERE id_cliente = p_id;
  END;

  PROCEDURE insertar(
    p_nombres IN VARCHAR2, p_apellidos IN VARCHAR2,
    p_correo IN VARCHAR2, p_telefono IN VARCHAR2, p_id OUT NUMBER
  ) AS
  BEGIN
    INSERT INTO clientes(nombres, apellidos, correo, telefono)
    VALUES(TRIM(p_nombres), TRIM(p_apellidos), LOWER(TRIM(p_correo)), TRIM(p_telefono))
    RETURNING id_cliente INTO p_id;
  END;

  PROCEDURE actualizar(
    p_id IN NUMBER, p_nombres IN VARCHAR2, p_apellidos IN VARCHAR2,
    p_correo IN VARCHAR2, p_telefono IN VARCHAR2, p_filas OUT NUMBER
  ) AS
  BEGIN
    UPDATE clientes
       SET nombres = TRIM(p_nombres),
           apellidos = TRIM(p_apellidos),
           correo = LOWER(TRIM(p_correo)),
           telefono = TRIM(p_telefono)
     WHERE id_cliente = p_id;
    p_filas := SQL%ROWCOUNT;
  END;

  PROCEDURE eliminar(p_id IN NUMBER, p_filas OUT NUMBER) AS
  BEGIN
    DELETE FROM clientes WHERE id_cliente = p_id;
    p_filas := SQL%ROWCOUNT;
  END;
END pkg_clientes;

Backend/Security/PasswordHasher.cs
El hash usa PBKDF2-HMAC-SHA256, salt aleatorio y un formato versionado.
La API vuelve a calcular el hash con el salt almacenado y compara en
tiempo constante.
using System.Security.Cryptography;

namespace FullStackClientesOracle19c.Security;

public static class PasswordHasher
{
    private const int Iterations = 310_000;
    private const int SaltSize = 16;
    private const int KeySize = 32;

    public static string Hash(string password)
    {
        byte[] salt = RandomNumberGenerator.GetBytes(SaltSize);
        byte[] key = Rfc2898DeriveBytes.Pbkdf2(
            password, salt, Iterations, HashAlgorithmName.SHA256, KeySize);

        return $"pbkdf2-sha256${Iterations}${Convert.ToBase64String(salt)}${Convert.ToBase64String(key)}";
    }

    public static bool Verify(string password, string stored)
    {
        try
        {
            var parts = stored.Split('$');
            if (parts.Length != 4 || parts[0] != "pbkdf2-sha256") return false;

            int iterations = int.Parse(parts[1]);
            if (iterations < 100_000 || iterations > 2_000_000) return false;

            byte[] salt = Convert.FromBase64String(parts[2]);
            byte[] expected = Convert.FromBase64String(parts[3]);
            byte[] actual = Rfc2898DeriveBytes.Pbkdf2(
                password, salt, iterations, HashAlgorithmName.SHA256, expected.Length);

            return CryptographicOperations.FixedTimeEquals(actual, expected);
        }
        catch
        {
            return false;
        }
    }
}

Backend/Services/JwtTokenService.cs
using System.IdentityModel.Tokens.Jwt;
using System.Security.Claims;
using System.Text;
using FullStackClientesOracle19c.Dtos;
using Microsoft.IdentityModel.Tokens;

namespace FullStackClientesOracle19c.Services;

public sealed class JwtTokenService(IConfiguration config)
{
    public LoginResponse Create(int idUsuario, string usuario)
    {
        string secret = config["Jwt:Secret"]
            ?? throw new InvalidOperationException("Falta Jwt:Secret.");
        var key = new SymmetricSecurityKey(Encoding.UTF8.GetBytes(secret));
        var credentials = new SigningCredentials(key, SecurityAlgorithms.HmacSha256);
        int minutes = int.TryParse(config["Jwt:ExpirationMinutes"], out var m) ? m : 30;
        DateTime expires = DateTime.UtcNow.AddMinutes(minutes);

        var claims = new[]
        {
            new Claim(ClaimTypes.NameIdentifier, idUsuario.ToString()),
            new Claim(ClaimTypes.Name, usuario)
        };

        var token = new JwtSecurityToken(
            issuer: config["Jwt:Issuer"],
            audience: config["Jwt:Audience"],
            claims: claims,
            notBefore: DateTime.UtcNow,
            expires: expires,
            signingCredentials: credentials);

        return new LoginResponse(new JwtSecurityTokenHandler().WriteToken(token), usuario, expires);
    }
}

Backend/Repositories/ClienteRepository.cs
using System.Data;
using FullStackClientesOracle19c.Data;
using FullStackClientesOracle19c.Dtos;
using Oracle.ManagedDataAccess.Client;

namespace FullStackClientesOracle19c.Repositories;

public sealed class ClienteRepository(OracleConnectionFactory factory)
{
    private static OracleParameter In(string name, OracleDbType type, object? value, int size = 0)
    {
        var p = new OracleParameter(name, type) { Direction = ParameterDirection.Input, Value = value ?? DBNull.Value };
        if (size > 0) p.Size = size;
        return p;
    }

    private static OracleParameter Out(string name, OracleDbType type, int size = 0)
    {
        var p = new OracleParameter(name, type) { Direction = ParameterDirection.Output };
        if (size > 0) p.Size = size;
        return p;
    }

    private static ClienteResponse Map(OracleDataReader r) => new(
        Convert.ToInt32(r["ID_CLIENTE"]),
        r["NOMBRES"].ToString()!,
        r["APELLIDOS"].ToString()!,
        r["CORREO"].ToString()!,
        r["TELEFONO"] == DBNull.Value ? null : r["TELEFONO"].ToString(),
        Convert.ToDateTime(r["FECHA_REGISTRO"]));

    public async Task<List<ClienteResponse>> Listar()
    {
        await using var cn = factory.Create();
        await cn.OpenAsync();
        await using var cmd = new OracleCommand("pkg_clientes.listar", cn)
        { CommandType = CommandType.StoredProcedure };
        cmd.Parameters.Add("p_resultado", OracleDbType.RefCursor).Direction = ParameterDirection.Output;
        await using var r = await cmd.ExecuteReaderAsync();
        var list = new List<ClienteResponse>();
        while (await r.ReadAsync()) list.Add(Map(r));
        return list;
    }

    public async Task<ClienteResponse?> Obtener(int id)
    {
        await using var cn = factory.Create();
        await cn.OpenAsync();
        await using var cmd = new OracleCommand("pkg_clientes.obtener", cn)
        { CommandType = CommandType.StoredProcedure };
        cmd.Parameters.Add(In("p_id", OracleDbType.Int32, id));
        cmd.Parameters.Add("p_resultado", OracleDbType.RefCursor).Direction = ParameterDirection.Output;
        await using var r = await cmd.ExecuteReaderAsync();
        return await r.ReadAsync() ? Map(r) : null;
    }

    public async Task<int> Insertar(ClienteRequest x)
    {
        await using var cn = factory.Create();
        await cn.OpenAsync();
        await using var cmd = new OracleCommand("pkg_clientes.insertar", cn)
        { CommandType = CommandType.StoredProcedure };
        cmd.Parameters.Add(In("p_nombres", OracleDbType.Varchar2, x.Nombres, 100));
        cmd.Parameters.Add(In("p_apellidos", OracleDbType.Varchar2, x.Apellidos, 100));
        cmd.Parameters.Add(In("p_correo", OracleDbType.Varchar2, x.Correo, 150));
        cmd.Parameters.Add(In("p_telefono", OracleDbType.Varchar2, x.Telefono, 30));
        cmd.Parameters.Add(Out("p_id", OracleDbType.Int32));
        await cmd.ExecuteNonQueryAsync();
        return Convert.ToInt32(cmd.Parameters["p_id"].Value.ToString());
    }

    public async Task<bool> Actualizar(int id, ClienteRequest x)
    {
        await using var cn = factory.Create();
        await cn.OpenAsync();
        await using var cmd = new OracleCommand("pkg_clientes.actualizar", cn)
        { CommandType = CommandType.StoredProcedure };
        cmd.Parameters.Add(In("p_id", OracleDbType.Int32, id));
        cmd.Parameters.Add(In("p_nombres", OracleDbType.Varchar2, x.Nombres, 100));
        cmd.Parameters.Add(In("p_apellidos", OracleDbType.Varchar2, x.Apellidos, 100));
        cmd.Parameters.Add(In("p_correo", OracleDbType.Varchar2, x.Correo, 150));
        cmd.Parameters.Add(In("p_telefono", OracleDbType.Varchar2, x.Telefono, 30));
        cmd.Parameters.Add(Out("p_filas", OracleDbType.Int32));
        await cmd.ExecuteNonQueryAsync();
        return Convert.ToInt32(cmd.Parameters["p_filas"].Value.ToString()) > 0;
    }

    public async Task<bool> Eliminar(int id)
    {
        await using var cn = factory.Create();
        await cn.OpenAsync();
        await using var cmd = new OracleCommand("pkg_clientes.eliminar", cn)
        { CommandType = CommandType.StoredProcedure };
        cmd.Parameters.Add(In("p_id", OracleDbType.Int32, id));
        cmd.Parameters.Add(Out("p_filas", OracleDbType.Int32));
        await cmd.ExecuteNonQueryAsync();
        return Convert.ToInt32(cmd.Parameters["p_filas"].Value.ToString()) > 0;
    }
}

Backend/Controllers/AuthController.cs
using System.Data;
using FullStackClientesOracle19c.Data;
using FullStackClientesOracle19c.Dtos;
using FullStackClientesOracle19c.Security;
using FullStackClientesOracle19c.Services;
using Microsoft.AspNetCore.Mvc;
using Oracle.ManagedDataAccess.Client;

namespace FullStackClientesOracle19c.Controllers;

[ApiController]
[Route("api/auth")]
public sealed class AuthController(
    OracleConnectionFactory factory, JwtTokenService tokens) : ControllerBase
{
    [HttpPost("login")]
    public async Task<IActionResult> Login(LoginRequest request)
    {
        await using var cn = factory.Create();
        await cn.OpenAsync();

        // Se obtiene el hash almacenado para validar PBKDF2 en la aplicación.
        // El paquete PL/SQL valida la identidad/estado y la coincidencia del hash.
        string? storedHash = null;
        await using (var find = new OracleCommand(
            "SELECT password_hash FROM usuarios WHERE UPPER(nombre_usuario)=UPPER(:u) AND activo='S'", cn))
        {
            find.Parameters.Add("u", OracleDbType.Varchar2, request.Usuario, ParameterDirection.Input);
            var value = await find.ExecuteScalarAsync();
            storedHash = value == null || value == DBNull.Value ? null : value.ToString();
        }

        if (storedHash == null || !PasswordHasher.Verify(request.Password, storedHash))
            return Unauthorized(new { mensaje = "Usuario o contraseña incorrectos." });

        await using var cmd = new OracleCommand("pkg_seguridad.validar_usuario", cn)
        { CommandType = CommandType.StoredProcedure };
        cmd.Parameters.Add("p_nombre_usuario", OracleDbType.Varchar2, request.Usuario, ParameterDirection.Input);
        cmd.Parameters.Add("p_password_hash", OracleDbType.Varchar2, storedHash, ParameterDirection.Input);
        cmd.Parameters.Add("p_id_usuario", OracleDbType.Int32).Direction = ParameterDirection.Output;
        cmd.Parameters.Add("p_nombre", OracleDbType.Varchar2, 60).Direction = ParameterDirection.Output;
        cmd.Parameters.Add("p_estado", OracleDbType.Int32).Direction = ParameterDirection.Output;
        cmd.Parameters.Add("p_mensaje", OracleDbType.Varchar2, 200).Direction = ParameterDirection.Output;
        await cmd.ExecuteNonQueryAsync();

        if (Convert.ToInt32(cmd.Parameters["p_estado"].Value.ToString()) != 1)
            return Unauthorized(new { mensaje = "Usuario o contraseña incorrectos." });

        int id = Convert.ToInt32(cmd.Parameters["p_id_usuario"].Value.ToString());
        string nombre = cmd.Parameters["p_nombre"].Value.ToString()!;
        return Ok(tokens.Create(id, nombre));
    }
}

Backend/Controllers/ClientesController.cs
using FullStackClientesOracle19c.Dtos;
using FullStackClientesOracle19c.Repositories;
using Microsoft.AspNetCore.Authorization;
using Microsoft.AspNetCore.Mvc;
using Oracle.ManagedDataAccess.Client;

namespace FullStackClientesOracle19c.Controllers;

[ApiController]
[Authorize]
[Route("api/clientes")]
public sealed class ClientesController(ClienteRepository repo) : ControllerBase
{
    [HttpGet]
    public async Task<IActionResult> Listar()
    {
        try { return Ok(await repo.Listar()); }
        catch (OracleException) { return Problem("Error consultando Oracle.", statusCode: 500); }
    }

    [HttpGet("{id:int}")]
    public async Task<IActionResult> Obtener(int id)
    {
        if (id <= 0) return BadRequest(new { mensaje = "Identificador inválido." });
        try
        {
            var cliente = await repo.Obtener(id);
            return cliente is null ? NotFound() : Ok(cliente);
        }
        catch (OracleException) { return Problem("Error consultando Oracle.", statusCode: 500); }
    }

    [HttpPost]
    public async Task<IActionResult> Crear(ClienteRequest request)
    {
        try
        {
            int id = await repo.Insertar(request);
            var creado = await repo.Obtener(id);
            return CreatedAtAction(nameof(Obtener), new { id }, creado);
        }
        catch (OracleException ex) when (ex.Number == 1)
        { return Conflict(new { mensaje = "Ya existe un cliente con ese correo." }); }
        catch (OracleException) { return Problem("No fue posible registrar el cliente.", statusCode: 500); }
    }

    [HttpPut("{id:int}")]
    public async Task<IActionResult> Actualizar(int id, ClienteRequest request)
    {
        if (id <= 0) return BadRequest();
        try
        {
            bool actualizado = await repo.Actualizar(id, request);
            return actualizado ? NoContent() : NotFound();
        }
        catch (OracleException ex) when (ex.Number == 1)
        { return Conflict(new { mensaje = "Ya existe un cliente con ese correo." }); }
        catch (OracleException) { return Problem("No fue posible actualizar el cliente.", statusCode: 500); }
    }

    [HttpDelete("{id:int}")]
    public async Task<IActionResult> Eliminar(int id)
    {
        if (id <= 0) return BadRequest();
        try { return await repo.Eliminar(id) ? NoContent() : NotFound(); }
        catch (OracleException) { return Problem("No fue posible eliminar el cliente.", statusCode: 500); }
    }
}

Frontend/app.js
const API = "https://localhost:7236/api";
let token = null;
let clientesCache = [];

const $ = (id) => document.getElementById(id);
const mensaje = (texto, error = false) => {
  $("mensaje").textContent = texto;
  $("mensaje").style.color = error ? "#b91c1c" : "#15803d";
};

async function request(ruta, opciones = {}) {
  const headers = { "Content-Type": "application/json", ...(opciones.headers || {}) };
  if (token) headers.Authorization = `Bearer ${token}`;

  const respuesta = await fetch(`${API}${ruta}`, { ...opciones, headers });
  if (respuesta.status === 401 && ruta !== "/auth/login") {
    cerrarSesion();
    throw new Error("Sesión vencida. Inicia sesión nuevamente.");
  }
  if (!respuesta.ok) {
    let detalle = `Error HTTP ${respuesta.status}`;
    try {
      const data = await respuesta.json();
      detalle = data.mensaje || data.title || detalle;
    } catch {}
    throw new Error(detalle);
  }
  if (respuesta.status === 204) return null;
  return respuesta.json();
}

$("loginForm").addEventListener("submit", async (e) => {
  e.preventDefault();
  try {
    const data = await request("/auth/login", {
      method: "POST",
      body: JSON.stringify({ usuario: $("usuario").value.trim(), password: $("password").value })
    });
    token = data.token;
    $("loginPanel").classList.add("oculto");
    $("appPanel").classList.remove("oculto");
    $("password").value = "";
    mensaje(`Bienvenido, ${data.usuario}.`);
    await cargarClientes();
  } catch (err) { mensaje(err.message, true); }
});

async function cargarClientes() {
  try {
    clientesCache = await request("/clientes");
    pintarTabla();
    mensaje(`${clientesCache.length} cliente(s) consultado(s).`);
  } catch (err) { mensaje(err.message, true); }
}

function pintarTabla() {
  const tbody = $("clientesBody");
  tbody.replaceChildren();

  for (const c of clientesCache) {
    const tr = document.createElement("tr");
    const valores = [c.idCliente, c.nombres, c.apellidos, c.correo, c.telefono || ""];
    for (const valor of valores) {
      const td = document.createElement("td");
      td.textContent = valor;
      tr.appendChild(td);
    }

    const acciones = document.createElement("td");
    const editar = document.createElement("button");
    editar.textContent = "Editar";
    editar.addEventListener("click", () => editarCliente(c.idCliente));

    const eliminar = document.createElement("button");
    eliminar.textContent = "Eliminar";
    eliminar.className = "btn-eliminar";
    eliminar.addEventListener("click", () => eliminarCliente(c.idCliente));

    acciones.append(editar, eliminar);
    tr.appendChild(acciones);
    tbody.appendChild(tr);
  }
}

$("clienteForm").addEventListener("submit", async (e) => {
  e.preventDefault();
  const id = $("idCliente").value;
  const cliente = {
    nombres: $("nombres").value.trim(),
    apellidos: $("apellidos").value.trim(),
    correo: $("correo").value.trim(),
    telefono: $("telefono").value.trim() || null
  };

  if (!cliente.nombres || !cliente.apellidos || !cliente.correo) {
    mensaje("Completa los campos obligatorios.", true);
    return;
  }

  try {
    await request(id ? `/clientes/${id}` : "/clientes", {
      method: id ? "PUT" : "POST",
      body: JSON.stringify(cliente)
    });
    limpiarFormulario();
    await cargarClientes();
    mensaje(id ? "Cliente actualizado." : "Cliente registrado.");
  } catch (err) { mensaje(err.message, true); }
});

function editarCliente(id) {
  const c = clientesCache.find(x => x.idCliente === id);
  if (!c) return;
  $("idCliente").value = c.idCliente;
  $("nombres").value = c.nombres;
  $("apellidos").value = c.apellidos;
  $("correo").value = c.correo;
  $("telefono").value = c.telefono || "";
  $("formTitle").textContent = `Editar cliente #${id}`;
  $("guardarBtn").textContent = "Actualizar cliente";
  window.scrollTo({ top: 0, behavior: "smooth" });
}

async function eliminarCliente(id) {
  if (!confirm(`¿Eliminar el cliente ${id}?`)) return;
  try {
    await request(`/clientes/${id}`, { method: "DELETE" });
    await cargarClientes();
    mensaje("Cliente eliminado.");
  } catch (err) { mensaje(err.message, true); }
}

function limpiarFormulario() {
  $("clienteForm").reset();
  $("idCliente").value = "";
  $("formTitle").textContent = "Registrar cliente";
  $("guardarBtn").textContent = "Guardar cliente";
}

function cerrarSesion() {
  token = null;
  clientesCache = [];
  $("appPanel").classList.add("oculto");
  $("loginPanel").classList.remove("oculto");
  $("clientesBody").replaceChildren();
  limpiarFormulario();
}

$("cancelarBtn").addEventListener("click", limpiarFormulario);
$("recargarBtn").addEventListener("click", cargarClientes);
$("logoutBtn").addEventListener("click", cerrarSesion);

13. Consideraciones importantes de seguridad
- El ejemplo conserva el JWT en memoria JavaScript; al recargar la
  página se debe iniciar sesión de nuevo. No se guarda el token en
  localStorage.
- Para producción, considera cookies HttpOnly, Secure y
  SameSite, protección CSRF y una estrategia de
  renovación/revocación de tokens.
- Añade rate limiting y bloqueo temporal ante intentos repetidos de
  inicio de sesión.
- El usuario Oracle de la API debe tener permisos mínimos sobre las
  tablas y paquetes; nunca uses SYSTEM.
- El ejemplo consulta el hash para verificar PBKDF2 en C# y luego
  llama al procedimiento PL/SQL pkg_seguridad.validar_usuario. Las
  consultas CRUD sí se ejecutan mediante procedimientos almacenados.
- Restringe AllowedOrigins a los dominios reales del frontend. No
  uses AllowAnyOrigin con autenticación.
- No expongas mensajes internos, cadenas de conexión, hashes o trazas
  de Oracle a los clientes HTTP.
- Configura HTTPS en todos los entornos y almacena los secretos fuera
  del código fuente.
